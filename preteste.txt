def bubbleSort(lst):
    n = len(lst)
    for i in range(n):
        for j in range(n - 1 - i):
            if lst[j] > lst[j + 1]:
                lst[j], lst[j + 1] = lst[j + 1], lst[j]
    return lst
2. Função InsertionSort

def insertionSort(lst):
    for i in range(1, len(lst)):
        key = lst[i]
        j = i - 1
        while j >= 0 and key < lst[j]:
            lst[j + 1] = lst[j]
            j -= 1
        lst[j + 1] = key
    return lst
3. Função ProcuraBinaria (Busca Binária)
python
Copy code
def procuraBinaria(lst, target):
    left, right = 0, len(lst) - 1
    while left <= right:
        mid = (left + right) // 2
        if lst[mid] == target:
            return mid
        elif lst[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1  # Retorna -1 se não encontrar

Código:

# Estrutura inicial do grafo
social_network = {
    "Filipe": {"Luis", "Americo"},
    "Luis": {"Filipe", "Sofia"},
    "Americo": {"Filipe"},
    "Pedro": {"Rita", "Sofia"},
    "Rita": {"Pedro"},
    "Sara": {"Sofia"},
    "Sofia": {"Sara", "Luis", "Pedro"}
}

# 1. Determinar o número de amigos de um utilizador
def friendNumber(user):
    return len(social_network.get(user, set()))

# 2. Verificar se um utilizador é amigo de outro
def isFriendOf(user1, user2):
    return user2 in social_network.get(user1, set())

# 3. Adicionar um amigo à lista de amigos de um utilizador
def addFriend(user1, user2):
    social_network.setdefault(user1, set()).add(user2)
    social_network.setdefault(user2, set()).add(user1)

# 4. Remover um amigo da lista de amigos de um utilizador
def removeFriend(user1, user2):
    social_network.get(user1, set()).discard(user2)
    social_network.get(user2, set()).discard(user1)

# 5. Popule a estrutura com 10 utilizadores e 20 conexões
def populateSocialNetwork():
    new_users = ["Ana", "Carlos", "Mariana", "João"]
    new_connections = [
        ("Ana", "Carlos"), ("Ana", "Luis"), ("João", "Pedro"),
        ("Carlos", "Mariana"), ("Mariana", "João"),
        ("Ana", "Rita"), ("Mariana", "Rita"),
        ("Sara", "Ana"), ("Carlos", "Pedro"), ("João", "Filipe")
    ]
    for user in new_users:
        social_network.setdefault(user, set())
    for user1, user2 in new_connections:
        addFriend(user1, user2)

# 6. Determinar se existe uma conexão entre A e B
def existsConnection(user1, user2, visited=None):
    if visited is None:
        visited = set()
    if user1 not in social_network or user2 not in social_network:
        return False
    if user2 in social_network[user1]:
        return True
    visited.add(user1)
    for friend in social_network[user1]:
        if friend not in visited:
            if existsConnection(friend, user2, visited):
                return True
    return False

# Estruturas da máquina
bebidas = {"Coca-Cola": 1.5, "Água": 1.0, "Café": 0.8}
moedas = {0.01: 10, 0.1: 10, 0.5: 5, 1: 10, 2: 5}
estado = "Standby"
saldo = 0

# 1. Inserir moeda
def inserirMoeda(valor):
    global saldo, estado
    if valor in moedas:
        saldo += valor
        moedas[valor] += 1
        estado = "Moedas"
        print(f"Saldo atual: {saldo:.2f}€")
    else:
        print("Moeda inválida.")

# 2. Selecionar bebida
def selecionarBebida(nome):
    global saldo, estado
    if nome not in bebidas:
        print("Bebida indisponível.")
        return
    preco = bebidas[nome]
    if saldo < preco:
        estado = "Insuficiente"
        print(f"Saldo insuficiente. Faltam {preco - saldo:.2f}€.")
    else:
        saldo -= preco
        estado = "Emitido"
        print(f"{nome} emitida. Troco: {saldo:.2f}€.")
        saldo = 0  # Zerar saldo após entrega

# 3. Cancelar operação
def cancelar():
    global saldo, estado
    print(f"Operação cancelada. Troco devolvido: {saldo:.2f}€.")
    saldo = 0
    estado = "Standby"

# 4. Menu da máquina
def menu():
    global estado
    while True:
        print(f"\nEstado atual: {estado}")
        print("1. Inserir moeda")
        print("2. Selecionar bebida")
        print("3. Cancelar")
        print("4. Sair")
        opcao = input("Escolha uma opção: ")
        if opcao == "1":
            valor = float(input("Digite o valor da moeda: "))
            inserirMoeda(valor)
        elif opcao == "2":
            bebida = input("Digite o nome da bebida: ")
            selecionarBebida(bebida)
        elif opcao == "3":
            cancelar()
        elif opcao == "4":
            print("Encerrando máquina.")
            break
        else:
            print("Opção inválida.")
